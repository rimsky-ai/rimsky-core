# Dashboard & Observability Implementation Plan

**Goal:** Implement the three public observability protocols (Rimsky observability API on `rimsky-control-api`, executor observability protocol per executor, store observability protocol per store), the reference dashboard at `dashboards/rimsky-dashboard/`, conformance extensions, and the supporting tests, per `docs/specs/2026-05-02-dashboard-and-observability-design.md`.

**Architecture:** Three new public protocols share a generic `Event`/`TraceEvent` envelope with a small standard vocabulary plus free-form fallback. Rimsky observability is HTTP/JSON on control-api at `/v1/observability/*`; executor and store observability are gRPC + HTTP+JSON bridges, dialed by control-api at startup via a best-effort handshake (separate from the existing fail-fast dispatch handshake). Reference dashboard is a separate React+Vite+TS SPA + Hono Node server in `dashboards/rimsky-dashboard/`, deliberately decoupled from `core/` (no Go imports, no DB credentials, no internal channels — pure HTTP client of the public APIs).

**Tech Stack:** Go 1.21+ (existing), gRPC + protobuf, chi for HTTP routing, pgx for Postgres, testcontainers-go for integration tests, golangci-lint for linting; React 18 + Vite + TypeScript + Tailwind + shadcn/ui + TanStack Query + Hono on Node 22.6+ for the dashboard (22.6+ required for `--experimental-strip-types` and `import.meta.dirname`); vitest + Playwright for dashboard tests.

**Spec:** `docs/specs/2026-05-02-dashboard-and-observability-design.md` — read this in full before starting. The tasks below assume the spec is the source of truth for endpoint shapes, vocabulary, and contract semantics; this plan covers *how* to implement, not *what* to implement (the spec has that).

**Pre-v1 stage:** Per `.claude/rules/rules.md`, this codebase is pre-v1. Migrations are not added in this plan (the spec is read-only over existing schema). No backwards-compat shims; if a refactor is cleaner, take the cleaner path.

```proto
syntax = "proto3";

package rimsky.v1;

option go_package = "github.com/fallguy/rimsky/proto/v1/gen;genv1";

import "google/protobuf/timestamp.proto";
import "google/protobuf/struct.proto";

// ExecutorObservability is the optional read-only protocol every
// executor MAY implement to expose per-dispatch traces to dashboards.
// The dispatch protocol (node_executor.proto) is unchanged. See
// docs/specs/2026-05-02-dashboard-and-observability-design.md §2.
service ExecutorObservability {
  rpc GetCapabilities(GetCapabilitiesRequest) returns (ObservabilityCapabilities);
  rpc GetTrace(GetTraceRequest) returns (Trace);
  rpc StreamTrace(StreamTraceRequest) returns (stream TraceEvent);
}

message GetCapabilitiesRequest {}

message ObservabilityCapabilities {
  bool supports_trace_get = 1;
  bool supports_trace_stream = 2;
  uint64 retention_after_terminal_seconds = 3;
  CustomUI custom_ui = 4;
}

message CustomUI {
  string ui_url = 1;
  EmbedMode embed_mode = 2;
  string dispatch_url_template = 3;  // executor: {dispatch_id}|{instance_id}|{node_type}
}

enum EmbedMode {
  EMBED_MODE_UNSPECIFIED = 0;
  LINK = 1;
  IFRAME = 2;
  BOTH = 3;
}

message GetTraceRequest {
  string dispatch_id = 1;
}

message StreamTraceRequest {
  string dispatch_id = 1;
}

message Trace {
  string dispatch_id = 1;
  bool evicted = 2;
  bool complete = 3;
  repeated TraceEvent events = 4;
}

message TraceEvent {
  string event_id = 1;
  string parent_event_id = 2;
  google.protobuf.Timestamp timestamp = 3;
  Severity severity = 4;
  string category = 5;            // standard vocab token OR free-form
  string message = 6;
  google.protobuf.Struct attributes = 7;
}

enum Severity {
  SEVERITY_UNSPECIFIED = 0;
  DEBUG = 1;
  INFO = 2;
  WARN = 3;
  ERROR = 4;
}
```

```proto
package rimsky.v1;

option go_package = "github.com/fallguy/rimsky/proto/v1/gen;genv1";

import "google/protobuf/struct.proto";
import "executor_observability.proto";  // for CustomUI, EmbedMode, Severity

// StoreObservability is the optional read-only protocol every store MAY
// implement to expose per-claim views (state, history, store-chosen
// payload/address/region disclosure) plus optional admin views to
// dashboards. The dispatch protocol (store_service.proto) is unchanged.

  rpc GetClaim(GetClaimRequest) returns (ClaimDetail);
  rpc StreamClaim(StreamClaimRequest) returns (stream ClaimEvent);
  rpc ListClaims(ListClaimsRequest) returns (ClaimList);
  rpc GetAdminView(GetAdminViewRequest) returns (AdminView);
}

message GetStoreCapabilitiesRequest {}

message StoreObservabilityCapabilities {
  bool supports_claim_get = 1;
  bool supports_claim_stream = 2;
  bool supports_list_claims = 3;
  uint64 retention_after_terminal_seconds = 4;

}

message AdminViewDecl {
  string name = 1;
  string title = 2;
  string description = 3;
  repeated AdminViewParam params = 4;
}

message AdminViewParam {
  string name = 1;
  string type = 2;             // "string" | "int" | "bool"
  string description = 3;
  bool required = 4;
}

message GetClaimRequest {
  string claim_id = 1;
}

message StreamClaimRequest {
  string claim_id = 1;
}

message ClaimDetail {
  string claim_id = 1;

  google.protobuf.Struct address = 3;          // store's view; may be null
  google.protobuf.Struct payload = 4;          // store's view; may be null
  google.protobuf.Struct region = 5;           // store's view; may be null

  google.protobuf.Timestamp closed_at = 7;
  repeated ClaimEvent history = 8;

  OPEN = 1;
  COMMITTED = 2;
  ABANDONED = 3;
  RELEASED = 4;
  UNKNOWN = 5;
}
```

### Task C2 — Implement read endpoints (resource-oriented)

**Files:** `core/observability/handler.go` (and small per-resource files if it grows past ~500 lines per cold-read guidelines)

**Steps:**

1. Implement each endpoint listed in spec §1.2.1 through §1.2.6. For each:
   - Resolve path/query parameters.
   - Issue the SQL query against `pgxpool.Pool` (or `core/storage/postgres/` helpers if a fitting helper already exists).
   - Project the row(s) into the JSON shape documented in the spec.
   - Honor cursor pagination (default limit 50, max 500). Cursors are base64-encoded `{last_pk, last_ts}` JSON; opaque to clients.
   - Return errors as `{ "error": { "code", "message", "details" } }` with the appropriate HTTP status.
2. Endpoint inventory:
   - `GET /v1/observability/stores`, `GET /v1/observability/stores/{name}` — read declared stores from `cfg.Stores` + `Discovery.GetStoreCapabilities(name)` + `cfg.Stores[name].ObservabilityEndpoint` reachability status. For per-store detail, also read recent `rimsky_store_lifecycle` rows for the per-store delivery state.
   - `GET /v1/observability/executors`, `GET /v1/observability/executors/{name}` — same shape over `cfg.Executors`.
   - `GET /v1/observability/templates`, `GET /v1/observability/templates/{hash}` — query `rimsky_templates` + LEFT JOIN `rimsky_template_tags`. Detail includes deployed-instance summary (`SELECT id, instance_key, terminated_at FROM rimsky_instances WHERE template_hash = $1`).
   - `GET /v1/observability/instances`, `GET /v1/observability/instances/{id}` — query `rimsky_instances`; detail includes the cascade graph per spec §1.4 (see Task C3).
   - `GET /v1/observability/schedules` — query `rimsky_schedules` joined to `rimsky_nodes` for context.
   - `GET /v1/observability/frames`, `GET /v1/observability/frames/{id}` — query `rimsky_frames`; detail joins `rimsky_dispatch` for constituent dispatches.
   - `GET /v1/observability/nodes/{instance_id}/{node_type}` — query `rimsky_nodes WHERE instance_id = $1 AND node_type = $2`; include state, retry_counter, current_error_class, recent `rimsky_events` for the node, current `rimsky_dispatch` row if any, current `rimsky_lock_holders` rows the node holds (via `holder_node_id`).
   - `GET /v1/observability/dispatches` — list of currently-live dispatches (per spec §1.2.3 list endpoint). Filters: `state` (`pending`/`claimed`, derived from `claimed_by IS NULL`), `executor_name`, `instance_id` (joined via `node_id` → `rimsky_nodes`). Cursor by `enqueued_at DESC, id DESC`.
   - `GET /v1/observability/dispatches/{id}` — query `rimsky_dispatch WHERE id = $1`; honest 404 if the row was deleted (terminal).
   - `GET /v1/observability/lock-holders`, `GET /v1/observability/lock-holders/{id}` — query `rimsky_lock_holders`; detail joins `rimsky_claim_holders` for held subgraph. **Do not return `address` bytes** (spec §1.3).
   - `GET /v1/observability/events` — query `rimsky_events`; filters `instance_id`, `node_id`, `kind`, `kind_in` (comma-separated → `WHERE kind = ANY($n)`), `since` (timestamp). Cursor by descending `occurred_at`, tiebreaker `id DESC`.
   - `GET /v1/observability/system/health` — control-api liveness (always true if the process is responding); supervisor heartbeat status (from `rimsky_supervisors`: report each row's `id`, `last_heartbeat_at`, `active_node_count`, and a derived `healthy: NOW() - last_heartbeat_at < 5 * heartbeat_interval` per blessed-invariant 6); store/executor reachability (from `Discovery`); postgres connectivity (`SELECT 1`).
   - `GET /v1/observability/system/summary` — counts per spec §1.2.6: instances active vs. terminated, frames by `rimsky_frames.state`, currently-claimed dispatches (`COUNT(*) FROM rimsky_dispatch WHERE claimed_by IS NOT NULL`), terminal failures last hour (`COUNT(*) FROM rimsky_events WHERE kind = 'error' AND occurred_at > NOW() - INTERVAL '1 hour'`), active lock-holders.

The cold-read guideline of ~500 lines per file applies. Split per-resource handlers into separate files (`handler_templates.go`, `handler_instances.go`, etc.) if needed.

### Task C3 — Implement cascade graph computation

**Files:** `core/observability/cascade_graph.go`

**Steps:**

1. The cascade graph endpoint (`GET /v1/observability/instances/{id}` detail) needs to compute the node graph from the template spec's `graph` field. The template spec is stored as JSONB in `rimsky_templates.spec` for the row whose `id = rimsky_instances.template_hash`.
2. Read the spec JSONB; extract `graph` (a list of `{name, depends_on?: []}` declarations — exact shape per existing template-spec validators in `core/template/`).
3. For each node declaration, look up the `rimsky_nodes` row by `(instance_id, node_type)` (where `node_type` matches the template-graph node name).
4. Look up `last_terminal_event` per spec §1.4: `SELECT kind, occurred_at FROM rimsky_events WHERE node_id = $1 AND kind IN ('work_completed','error') ORDER BY occurred_at DESC LIMIT 1`.
5. Look up active dispatch: `SELECT id FROM rimsky_dispatch WHERE node_id = $1`.
6. Project to the shape documented in spec §1.4.

```ts
  url: string,
  onEvent: (event: T) => void,
  onComplete: () => void,
  onError?: (err: Error) => void,
): () => void {
  let es = new EventSource(url);
  let closed = false;
  es.onmessage = (e) => {
    try {
      const data = JSON.parse(e.data);
      if (data.category === 'trace_complete' || data.category === 'claim_terminal') {
        onComplete();
        es.close();
      } else {
        onEvent(data as T);
      }
    } catch (err) { onError?.(err as Error); }
  };
  es.onerror = () => {
    if (closed) return;
    es.close();
    setTimeout(() => { if (!closed) es = new EventSource(url); }, 1000);
  };
  return () => { closed = true; es.close(); };
}
```

```ts
import { config } from './config';

type PeerEndpoint = { name: string; observabilityHttpUrl: string };

let executors = new Map<string, PeerEndpoint>();
let stores = new Map<string, PeerEndpoint>();
let lastRefresh = 0;
const TTL = 30_000;

export async function getExecutorEndpoint(name: string): Promise<PeerEndpoint | null> {
  await refreshIfStale();
  if (executors.has(name)) return executors.get(name)!;
  // Cache is fresh-but-missing: force a refresh in case the peer was added since lastRefresh.
  await refresh();
  return executors.get(name) ?? null;
}
export async function getStoreEndpoint(name: string): Promise<PeerEndpoint | null> {
  await refreshIfStale();
  if (stores.has(name)) return stores.get(name)!;
  await refresh();
  return stores.get(name) ?? null;
}

async function refreshIfStale() {
  if (Date.now() - lastRefresh < TTL) return;
  await refresh();
}

async function refresh() {
  const [exs, sts] = await Promise.all([
    fetch(`${config.controlApiUrl}/v1/observability/executors`).then(r => r.json()),
    fetch(`${config.controlApiUrl}/v1/observability/stores`).then(r => r.json()),
  ]);
  executors = new Map((exs.executors ?? []).map((e: any) => [
    e.name, { name: e.name, observabilityHttpUrl: deriveObsUrl(e) },
  ]));
  stores = new Map((sts.stores ?? []).map((s: any) => [
    s.name, { name: s.name, observabilityHttpUrl: deriveObsUrl(s) },
  ]));
  lastRefresh = Date.now();
}

function deriveObsUrl(peer: any): string {
  // peer.observability_endpoint when present, else peer.endpoint; prefix http:// when missing scheme.
  const ep = peer.observability_endpoint ?? peer.endpoint;
  return ep.startsWith('http') ? ep : `http://${ep}`;
}
```

3. `proxy.ts` — Hono routes for `/api/control/*`, `/api/exec/:name/*`, `/api/store/:name/*`. SSE pass-through: detect `Accept: text/event-stream`, stream the upstream response chunks straight back to the client; do not buffer.

```ts
import { Hono } from 'hono';
import { stream } from 'hono/streaming';
import { config } from './config';
import { getExecutorEndpoint, getStoreEndpoint } from './discovery';

export const proxy = new Hono();

proxy.all('/api/control/*', async (c) => {
  const upstreamPath = c.req.path.replace('/api/control', '/v1/observability');
  const search = new URL(c.req.url).search;
  return forward(c, `${config.controlApiUrl}${upstreamPath}${search}`);
});

proxy.all('/api/exec/:name/*', async (c) => {
  const ep = await getExecutorEndpoint(c.req.param('name'));
  if (!ep) return c.json({ error: { code: 'NOT_FOUND', message: 'unknown executor' } }, 404);
  const tail = c.req.path.split(`/api/exec/${ep.name}`)[1];
  const search = new URL(c.req.url).search;
  return forward(c, `${ep.observabilityHttpUrl}/observability/v1${tail}${search}`);
});

proxy.all('/api/store/:name/*', async (c) => {
  const ep = await getStoreEndpoint(c.req.param('name'));
  if (!ep) return c.json({ error: { code: 'NOT_FOUND', message: 'unknown store' } }, 404);
  const tail = c.req.path.split(`/api/store/${ep.name}`)[1];
  const search = new URL(c.req.url).search;
  return forward(c, `${ep.observabilityHttpUrl}/observability/v1${tail}${search}`);
});

function copyHeaders(src: Headers): Headers {
  const out = new Headers(src);
  // Strip hop-by-hop / Node-managed headers; let undici set Host from the URL.
  out.delete('host');
  out.delete('content-length');
  out.delete('connection');
  return out;
}

async function forward(c: any, url: string) {
  const accept = c.req.header('Accept') ?? '';
  const method = c.req.method;
  const init: RequestInit & { duplex?: 'half' } = {
    method,
    headers: copyHeaders(c.req.raw.headers),
  };
  if (method !== 'GET' && method !== 'HEAD') {
    // Buffer the request body — Node's fetch streaming requires `duplex: 'half'`,
    // but proxy bodies in v1 are small JSON; buffer for simplicity and broad compat.
    init.body = await c.req.raw.arrayBuffer();
  }
  const upstream = await fetch(url, init);
  if (accept.includes('text/event-stream')) {
    return stream(c, async (s) => {
      const reader = upstream.body!.getReader();
      while (true) {
        const { value, done } = await reader.read();
        if (done) break;
        await s.write(value);
      }
    });
  }
  return new Response(upstream.body, { status: upstream.status, headers: upstream.headers });
}
```
